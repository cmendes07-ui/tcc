% --- Configuração Definitiva de Alíneas e Listas Pré-Textuais (Padrão UEMA) ---
\usepackage{enumitem}

\newlist{uemaenum}{enumerate}{1}
\setlist[uemaenum,1]{
    label=\alph*),
    leftmargin=2.75cm,
    labelsep=0.4cm,
    itemsep=2pt,
    topsep=2pt
}

% Para listas com marcadores (pontos), se precisar:
\newlist{uemaball}{itemize}{1}
\setlist[uemaball,1]{
    label=\textbullet,
    leftmargin=2.75cm,
    itemsep=2pt,
    topsep=2pt
}

% --- Títulos das Listas Pré-Textuais (Padrão UEMA) ---
\addto\captionsbrazil{%
  \renewcommand{\listfigurename}{LISTA DE ILUSTRAÇÕES}%
  \renewcommand{\listtablename}{LISTA DE TABELAS}%
  \renewcommand{\listadesiglasname}{LISTA DE SIGLAS}%
  \renewcommand{\listadesimbolosname}{LISTA DE SÍMBOLOS}%
}

% Ambiente para Lista de Abreviaturas (separada da Lista de Siglas conforme modelo UEMA)
\newcommand{\listadeabreviaturasname}{LISTA DE ABREVIATURAS}
\newenvironment{abreviaturas}{%
  \pretextualchapter{\listadeabreviaturasname}
  \begin{symbols}
}{%
  \end{symbols}
  \cleardoublepage
}
